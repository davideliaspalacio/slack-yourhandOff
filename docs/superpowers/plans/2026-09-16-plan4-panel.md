# Plan 4 — Panel web

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Un panel mínimo donde Anthony, Jake y David entren con su cuenta, vean a cada persona con su dossier, historial y costes, y puedan hacer tres cosas: cambiar el estado, volver a investigar y ajustar umbrales.

**Architecture:** Next.js en Vercel que lee Supabase directamente por PostgREST con la sesión del usuario, sin API propia. Todo el control de acceso vive en políticas RLS de Postgres: el panel no puede hacer nada que la base no le permita, aunque alguien manipule el navegador.

**Tech Stack:** Next.js (App Router) + TypeScript, `@supabase/supabase-js` y `@supabase/ssr`, Supabase Auth (enlace mágico por email), Vercel.

## Global Constraints

- **El panel nunca habla con Postgres directamente ni con la clave de servicio.** Solo la clave pública y la sesión del usuario.
- **RLS decide, no el frontend.** Lectura para usuarios autenticados; de escritura, solo las tres acciones.
- **Solo tres personas entran:** Anthony, Jake y David, invitados a mano en Supabase Auth. Nada de registro abierto.
- **El panel no edita dossiers.** Sin CRM, sin pipeline, sin edición de contenido.
- **Descartar es permanente y visible:** desde el panel se puede revivir, y queda registrado quién lo hizo.
- **Ningún secreto en el cliente:** las claves de OpenAI, Slack, Twilio o Resend no entran en este proyecto.

```sql
-- El panel entra como 'authenticated' con la sesión del usuario. Lectura de lo
-- que necesita ver; de escritura, solo las tres acciones del spec §11.

create policy panel_lee_prospects on prospects for select to authenticated using (true);
create policy panel_lee_dossiers on dossiers for select to authenticated using (true);
create policy panel_lee_mensajes on slack_messages for select to authenticated using (true);
create policy panel_lee_entregas on deliveries for select to authenticated using (true);
create policy panel_lee_acciones on agent_actions for select to authenticated using (true);
create policy panel_lee_llm on llm_calls for select to authenticated using (true);
create policy panel_lee_costes on cost_events for select to authenticated using (true);
create policy panel_lee_config on config for select to authenticated using (true);
create policy panel_lee_cola on research_jobs for select to authenticated using (true);

-- Acción 1: cambiar el estado. Solo entre estados que el panel puede fijar.
create policy panel_cambia_estado on prospects for update to authenticated
    using (true)
    with check (state in ('nuevo', 'investigado', 'contactado', 'cliente', 'descartado'));

-- Acción 2: volver a investigar. Solo tareas manuales, y la cola ya impide dos
-- abiertas por persona.
create policy panel_reinvestiga on research_jobs for insert to authenticated
    with check (reason = 'manual' and status = 'pendiente');

-- Acción 3: ajustar umbrales. Solo estas claves: el kill switch y los topes de
-- dinero se tocan a mano, no desde una pantalla.
create policy panel_ajusta_umbrales on config for update to authenticated
    using (key in ('banda_alta_min', 'banda_media_min', 'banda_baja_min',
                   'research_por_hora', 'sms_por_dia', 'sms_hora_inicio', 'sms_hora_fin'))
    with check (key in ('banda_alta_min', 'banda_media_min', 'banda_baja_min',
                        'research_por_hora', 'sms_por_dia', 'sms_hora_inicio', 'sms_hora_fin'));

-- Gasto por día, ya sumado: el panel no tiene por qué recorrer el ledger entero.
create view panel_costes_diarios as
select dia, sum(usd) as usd, sum(llamadas) as llamadas
from (
    select date(created_at) as dia, sum(cost_usd) as usd, count(*) as llamadas from llm_calls group by 1
    union all
    select date(created_at), sum(cost_usd), count(*) from cost_events group by 1
) t
group by dia
order by dia desc;
```

```python
# tests/test_panel_rls.py
"""Las políticas se prueban como las ve el panel: con el rol authenticated,
no con el superusuario, que se las salta todas."""
import pytest

from handoff_agent import db


def as_panel(sql, params=()):
    with db.transaction() as cur:
        cur.execute("set local role authenticated")
        cur.execute(sql, params)
        return cur.fetchall() if cur.description else None


def test_the_panel_can_read_people(conn):
    db.execute("insert into prospects (slack_user_id, full_name) values ('U1', 'Ada')")
    assert len(as_panel("select * from prospects")) == 1


def test_the_panel_can_discard_and_revive(conn):
    db.execute("insert into prospects (slack_user_id) values ('U1')")
    as_panel("update prospects set state = 'descartado' where slack_user_id = 'U1'")
    assert db.fetch_one("select state from prospects")["state"] == "descartado"


def test_the_panel_cannot_delete_anyone(conn):
    db.execute("insert into prospects (slack_user_id) values ('U1')")
    as_panel("delete from prospects where slack_user_id = 'U1'")
    assert db.fetch_one("select count(*) as n from prospects")["n"] == 1


def test_the_panel_can_only_queue_manual_jobs(conn):
    as_panel("insert into research_jobs (slack_user_id, reason) values ('U1', 'manual')")
    with pytest.raises(Exception):
        as_panel("insert into research_jobs (slack_user_id, reason) values ('U2', 'mensaje')")


def test_the_panel_cannot_touch_the_kill_switch(conn):
    as_panel("update config set value = 'true'::jsonb where key = 'kill_switch'")
    assert db.fetch_one("select value from config where key = 'kill_switch'")["value"] is False


def test_the_panel_cannot_write_dossiers(conn):
    db.execute("insert into prospects (slack_user_id) values ('U1')")
    pid = db.fetch_one("select id from prospects")["id"]
    with pytest.raises(Exception):
        as_panel("insert into dossiers (prospect_id, version, content, sources) "
                 "values (%s, 1, '{}'::jsonb, '[]'::jsonb)", (pid,))
```

```typescript
import { createServerClient } from "@supabase/ssr";
import { cookies } from "next/headers";

export async function createClient() {
  const store = await cookies();
  return createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        getAll: () => store.getAll(),
        setAll: (list) => list.forEach(({ name, value, options }) => store.set(name, value, options)),
      },
    },
  );
}
```

```tsx
// panel/app/login/page.tsx
"use client";
import { useState } from "react";
import { createClient } from "@/lib/supabase/client";

export default function Login() {
  const [email, setEmail] = useState("");
  const [sent, setSent] = useState(false);

  async function send(e: React.FormEvent) {
    e.preventDefault();
    await createClient().auth.signInWithOtp({
      email,
      options: { emailRedirectTo: `${location.origin}/personas` },
    });
    setSent(true); // No se distingue un email invitado de uno que no lo está.
  }

  if (sent) return <p>Si esa cuenta tiene acceso, recibirás un enlace para entrar.</p>;
  return (
    <form onSubmit={send}>
      <input type="email" value={email} onChange={(e) => setEmail(e.target.value)} required />
      <button type="submit">Entrar</button>
    </form>
  );
}
```

```typescript
// panel/middleware.ts
import { NextResponse, type NextRequest } from "next/server";
import { createServerClient } from "@supabase/ssr";

export async function middleware(request: NextRequest) {
  const response = NextResponse.next();
  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        getAll: () => request.cookies.getAll(),
        setAll: (list) => list.forEach(({ name, value, options }) => response.cookies.set(name, value, options)),
      },
    },
  );
  const { data } = await supabase.auth.getUser();
  if (!data.user && !request.nextUrl.pathname.startsWith("/login")) {
    return NextResponse.redirect(new URL("/login", request.url));
  }
  return response;
}

export const config = { matcher: ["/((?!_next/static|_next/image|favicon.ico).*)"] };
```

```tsx
// panel/app/(panel)/personas/page.tsx
import Link from "next/link";
import { createClient } from "@/lib/supabase/server";

export default async function Personas({ searchParams }: { searchParams: Promise<Record<string, string>> }) {
  const { estado, banda, desde } = await searchParams;
  const supabase = await createClient();
  let query = supabase
    .from("prospects")
    .select("id, full_name, company_name, state, updated_at, deliveries(band, created_at)")
    .order("updated_at", { ascending: false })
    .limit(200);
  if (estado) query = query.eq("state", estado);
  if (desde) query = query.gte("updated_at", desde);
  const { data: personas } = await query;

  const visibles = banda
    ? personas?.filter((p) => p.deliveries?.some((d) => d.band === banda))
    : personas;

  return (
    <table>
      <tbody>
        {visibles?.map((p) => (
          <tr key={p.id}>
            <td><Link href={`/personas/${p.id}`}>{p.full_name ?? "(sin nombre)"}</Link></td>
            <td>{p.company_name}</td>
            <td>{p.state}</td>
            <td>{p.deliveries?.[0]?.band ?? "—"}</td>
            <td>{new Date(p.updated_at).toLocaleDateString()}</td>
          </tr>
        ))}
      </tbody>
    </table>
  );
}
```

```typescript
"use server";
import { revalidatePath } from "next/cache";
import { createClient } from "@/lib/supabase/server";

export async function cambiarEstado(id: string, estado: string) {
  const supabase = await createClient();
  const { error } = await supabase.from("prospects").update({ state: estado }).eq("id", id);
  if (error) throw error; // RLS rechaza cualquier estado que no esté permitido.
  revalidatePath(`/personas/${id}`);
}

export async function volverAInvestigar(id: string, slackUserId: string) {
  const supabase = await createClient();
  // La cola solo admite una tarea abierta por persona: un segundo clic no
  // encola dos investigaciones, falla contra el índice y no pasa nada.
  await supabase.from("research_jobs").insert({ slack_user_id: slackUserId, reason: "manual" });
  revalidatePath(`/personas/${id}`);
}
```
